		always = no

	}

}

law_sharia = { #islamic flavor
	group = lawgroup_policing
	parent = law_religious_jurisprudence
	
	icon = "gfx/interface/icons/law_icons/religious_jurisdiction.dds"
	
	unlocking_technologies = {
		tech_bureaucracy
	}

	progressiveness = 50

	is_visible = {
		country_is_islamic = yes
	}

	requires_law_or = {
		law_theocracy
		law_subjecthood
		law_state_religion
	}

	modifier = {
		country_institution_cost_institution_police_mult = -0.2
	}

	institution = institution_police
	institution_modifier = {
		interest_group_ig_devout_pol_str_mult = 0.1
		state_turmoil_effects_mult = -0.05
		country_authority_add = -50

		state_conversion_mult = 0.05

		country_radicalism_increases_second_rate_citizen_mult = -0.01
		country_radicalism_increases_open_prejudice_mult = -0.05
		country_radicalism_increases_cultural_erasure_mult = -0.05
		country_radicalism_increases_violent_hostility_mult = -0.05
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	# can impose in religious or policing/security power blocs
	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				OR = {
					AND = {
						modifier:country_can_impose_same_lawgroup_policing_in_power_bloc_bool = yes
						has_law = scope:law
					}
					AND = {
						modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
						has_law = scope:law
					}
				}
				
			}
			can_impose_law_default = yes
		}
	}

	ai_impose_chance = {
		value = 0

		if = {
			limit = {
				has_law = law_type:law_state_religion
				power_bloc ?= { has_identity = identity:identity_religious }
				religion = scope:target_country.religion
			}

			add = base_impose_law_weight
		}

		if = {
			limit = {
				has_law = law_type:law_sharia
				religion = scope:target_country.religion
				power_bloc ?= { has_principle_group = principle_group:principle_group_police_coordination }
			}

			add = base_impose_law_weight
		}

	}
}

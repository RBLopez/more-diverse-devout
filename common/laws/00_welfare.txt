law_religious_charity = {
	group = lawgroup_welfare
	
	icon = "gfx/interface/icons/law_icons/religious_charity.dds"

	progressiveness = 25

	disallowing_laws = {
		law_total_separation
	}

	modifier = {
		country_institution_cost_institution_welfare_mult = -0.2
	}

	institution = institution_social_security
	institution_modifier = {
		state_dependent_wage_mult = 0.1
		state_food_security_add = 0.01
		state_welfare_payments_add = -0.2  #cancel default state action
	}

	can_enact = { #only allow this in countries with appropriate state religions
		owner.ig:ig_devout = {
			law_stance = {
				law = law_type:law_religious_charity
				value > neutral
			}
		}
	}

	ai_will_do = {
		owner.ig:ig_devout = {
			law_stance = {
				law = law_type:law_religious_charity
				value > neutral
			}
		}
		exists = ruler
		ruler = {
			OR = {
				has_ideology = ideology:ideology_theocrat
				interest_group = owner.ig:ig_devout
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

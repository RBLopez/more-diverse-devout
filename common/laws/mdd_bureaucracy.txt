law_religious_administration = {
	group = lawgroup_bureaucracy
	
	icon = "gfx/interface/icons/law_icons/religious_administration.dds"

	progressiveness = -50

	is_visible = { # only show for reference in places where the flavored variant is not available
		NOR = {
			country_has_state_religion = rel:sunni
			country_has_state_religion = rel:shiite
			country_has_state_religion = rel:ibadi
			c:CHI ?= this
			AND = {
				exists = c:CHI
				is_subject_of = c:CHI
				country_has_state_religion = rel:confucian
			}
		}
	}

	modifier = {
		state_bureaucracy_population_base_cost_factor_mult = -0.25
		interest_group_ig_devout_pol_str_mult = 0.25
	}

	disallowing_laws = {
		law_council_republic
		law_total_separation
	}

	can_enact = {
		always = no
		# use always = no here and only allow specific, religion-specific variants
	}
}

law_dhimmi = { #Islamic administration.
	group = lawgroup_bureaucracy
	parent = law_religious_administration
	
	icon = "gfx/interface/icons/law_icons/people_of_the_book.dds"

	progressiveness = -50

	is_visible = {
		country_is_islamic = yes
	}

	requires_law_or = {
		law_people_of_the_book
		law_millet_system
	}

	modifier = {
		state_bureaucracy_population_base_cost_factor_mult = -0.25
		interest_group_ig_devout_pol_str_mult = 0.25

		state_conscription_rate_mult = -0.4
	}

	tax_modifier_very_low = {
		tax_heathen_add = 0.4
	}

	tax_modifier_low = {
		tax_heathen_add = 0.45
	}

	tax_modifier_medium = {
		tax_heathen_add = 0.5
	}

	tax_modifier_high = {
		tax_heathen_add = 0.55
	}

	tax_modifier_very_high = {
		tax_heathen_add = 0.6
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_theocrat
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_imperial_exams = { # Confucian flavor
	group = lawgroup_bureaucracy
	parent = law_religious_administration
	
	icon = "gfx/interface/icons/law_icons/religious_administration.dds"

	progressiveness = -50

	is_visible = {
		OR = {
			c:CHI ?= this # For the Chinese Empire and its subjects
			AND = {
				exists = c:CHI
				is_subject_of = c:CHI
				country_has_state_religion = rel:confucian
			}
		}
	}

	modifier = {
		state_bureaucracy_population_base_cost_factor_mult = -0.25
		country_bureaucrats_pol_str_mult = 0.25
		country_authority_add = -75
	}

	requires_law_or = {
		law_theocracy
		law_monarchy
		law_chiefdom
		law_state_religion
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

REPLACE:law_appointed_bureaucrats = {
	group = lawgroup_bureaucracy
	
	icon = "gfx/interface/icons/law_icons/appointed_bureaucrats.dds"
	
	progressiveness = 0
	
	modifier = {
		country_bureaucrats_pol_str_mult = 0.25
		state_tax_capacity_mult = 0.25
	}


	on_enact = {
		if = {
			limit = {
				has_law = law_type:law_elected_bureaucrats
			}
			custom_tooltip = bureaucrats_pb_ig_shift_effect_negative_tt
		}
		if = {
			limit = {
				has_law = law_type:law_hereditary_bureaucrats
			}
			custom_tooltip = urban_aristocrats_int_ig_shift_effect_negative_tt
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = { 
				OR = {
					has_journal_entry = je_sick_man_bureaucracy
					has_journal_entry = je_great_reforms_bureaucratic
				}
				OR = {
					has_law = law_type:law_hereditary_bureaucrats
					has_law = law_type:law_religious_administration
				}
				
			}
			add = 1000
		}		
	}
}

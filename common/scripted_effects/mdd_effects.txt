#this effect assumes country scope
effect_create_minority_ig_leader = {
	if = {
		limit = {
			exists = var:minority_faith
			ig:ig_minority_faiths.leader = {
				is_historical = no
			}
		}
		random_scope_pop = {
			limit = {
				is_pop_type = clergymen
				trigger_if = {
					limit = {
						owner.var:minority_faith = rel:catholic
					}
					has_pop_religion = catholic
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:protestant
					}
					has_pop_religion = protestant
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:orthodox
					}
					has_pop_religion = orthodox
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:oriental_orthodox
					}
					has_pop_religion = oriental_orthodox
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:sunni
					}
					has_pop_religion = sunni
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:shiite
					}
					has_pop_religion = shiite
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:jewish
					}
					has_pop_religion = jewish
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:mahayana
					}
					has_pop_religion = mahayana
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:gelugpa
					}
					has_pop_religion = gelugpa
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:theravada
					}
					has_pop_religion = theravada
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:hindu
					}
					has_pop_religion = hindu
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:shinto
					}
					has_pop_religion = shinto
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:sikh
					}
					has_pop_religion = sikh
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:animist
					}
					has_pop_religion = animist
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:atheist
					}
					has_pop_religion = atheist
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:anglican
					}
					has_pop_religion = anglican
				}
				trigger_else_if = {
					limit = {
						owner.var:minority_faith = rel:confucian
					}
					has_pop_religion = confucian
				}
			}
			save_scope_as = clergypop
		}
		if = {
			limit = {
				exists = scope:clergypop
			}
			scope:clergypop.culture = {
				save_scope_as = clergy_culture
			}
			create_character = {
				culture = scope:clergy_culture
				religion = var:minority_faith
				interest_group = ig:ig_minority_faiths
				ig_leader = yes
				
			}
		}	
	}
}

#country scope
effect_set_manor_house_religious_pm = {
	if = {
		limit = {
			has_law = law_type:law_state_atheism
		}
		# no actions needed, pms sort themselves out
	}
	else_if = {
		limit = {
			OR = {
				ig:ig_devout = {
					has_ideology = ideology:ideology_church_lands
				}
				ig:ig_devout = {
					has_ideology = ideology:ideology_church_lands_russia
				}
				# Add special handling for East India Company and british indian colonies?
			}
		}
		every_scope_state = {
			if = {
				limit = {
					NOT = {
						is_production_method_active = {
							building_type = building_manor_house
							production_method = pm_manor_house_church_lands
						}
					}
				}
				activate_production_method = {
					building_type = building_manor_house
					production_method = pm_manor_house_church_lands
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				religion = {
					has_discrimination_trait = heritage_christian
				}
				religion = {
					has_discrimination_trait = heritage_islamic
				}
				religion = { #buddhism
					has_discrimination_trait = heritage_dharmic
				}
				religion = { #hinduism, sikhism
					has_discrimination_trait = heritage_indic
				}
			}
		}
		every_scope_state = {
			if = {
				limit = {
					NOT = {
						is_production_method_active = {
							building_type = building_manor_house
							production_method = pm_manor_house_temple_grounds
						}
					}
				}
				activate_production_method = {
					building_type = building_manor_house
					production_method = pm_manor_house_temple_grounds
				}
			}
		}
	}
	else = {
		every_scope_state = {
			if = {
				limit = {
					NOT = {
						is_production_method_active = {
							building_type = building_manor_house
							production_method = pm_manor_house_patronage
						}
					}
				}
				activate_production_method = {
					building_type = building_manor_house
					production_method = pm_manor_house_patronage
				}
			}
		}
	}
}

#country scope. sets correct religious administration law variant based on the state religion
effect_set_religious_administration_law = {
	if = {
		limit = {
			OR = {
				religion = rel:sunni
				religion = rel:shiite
				religion = rel:ibadi
			}
			NOT = { has_law = law_type:law_people_of_the_book }
		}
		activate_law = law_type:law_people_of_the_book	
	}
	else_if = {
		limit = {
			religion = rel:confucian
			NOT = { has_law = law_type:law_imperial_exams }
		}
		activate_law = law_type:law_imperial_exams
	}
}
